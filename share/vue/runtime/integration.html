const App = {

第六步：createApp 怎样组装 DOM Renderer 并首次挂载

Demo 执行 createApp(App) 时，runtime-dom 先取得一个注入了浏览器操作的 renderer，再调用这个 renderer 提供的应用工厂。renderer 在模块内复用，app、组件实例和 VNode 则按每次调用重新创建；这是两种不同的生命周期。

6.1 runtime-dom 合并宿主操作，并按需创建 renderer

nodeOps 提供创建、插入、删除和文本写入， patchProp 负责属性与事件分派。两者合成 rendererOptions。普通客户端入口调用 ensureRenderer()；SSR 水合入口调用 ensureHydrationRenderer()。 

/** core renderer 所需的完整 DOM 宿主选项。 */
const rendererOptions = /*@__PURE__*/ extend({ patchProp }, nodeOps)

// 惰性创建：若用户只从 Vue 导入响应式 API，core renderer 与全部 DOM 操作可被摇掉。
let renderer: Renderer<Element | ShadowRoot> | HydrationRenderer

/** 当前缓存 renderer 是否已升级为带 hydration 能力的版本。 */
let enabledHydration = false

/** 获取普通 DOM renderer，首次真正 render/createApp 时才创建。 */
function ensureRenderer() {
  return (
    renderer ||
    (renderer = createRenderer<Node, Element | ShadowRoot>(rendererOptions))
  )
}

/**
 * 获取水合 renderer。普通 renderer 已创建时会替换为 hydration 版本；两者宿主选项相同。
 */
function ensureHydrationRenderer() {
  renderer = enabledHydration
    ? renderer
    : createHydrationRenderer(rendererOptions)
  enabledHydration = true
  return renderer as HydrationRenderer
}
RENDERER CACHE · THREE VALID PATHS
普通客户端首次使用S0: renderer = undefined
→ ensureRenderer()
→ S1: 普通 renderer
首次就是 SSR 水合S0: undefined / false
→ ensureHydrationRenderer()
→ S2: 水合 renderer / true
普通 renderer 后来升级S1: 普通 renderer / false
→ 水合入口替换缓存
→ S2: 水合 renderer / true
升级之后再次客户端渲染ensureRenderer() 直接返回现有的 S2；不会降级， 也不会再造 renderer。

这里缓存的是包含算法与宿主操作的 renderer，不是 app。Demo 的 createApp(App) 每调用一次，后面的 createAppAPI 仍会创建新的 app context、插件集合和根实例。

6.4 同一个 Counter 中，谁长期持有谁

6.5 DOM 平台入口把容器交给 runtime-core

输入：'#app'。输出：组件公开代理。 状态变化：容器旧文本被清空，core mount 创建根 VNode 并调用 renderer 的 render。runtime-dom 的完整包装如下： 

/**
 * 创建客户端 DOM 应用，并扩展 mount：解析选择器、必要时读取 in-DOM template、清空
 * 容器、推导根命名空间，以及维护 v-cloak/data-v-app 标记。
 */
export const createApp = ((...args) => {
  const app = ensureRenderer().createApp(...args)

  if (__DEV__) {
    injectNativeTagCheck(app)
    injectCompilerOptionsCheck(app)
  }

  const { mount } = app
  /** 解析客户端 mount 容器；缺 render/template 时读取可信 in-DOM 模板，只清空 Element，保留 ShadowRoot 样式，再调用 core mount 并更新 v-cloak/data-v-app。 */
  app.mount = (containerOrSelector: Element | ShadowRoot | string): any => {
    const container = normalizeContainer(containerOrSelector)
    if (!container) return

    const component = app._component
    if (!isFunction(component) && !component.render && !component.template) {
      // __UNSAFE__：in-DOM template 会被编译并执行其中 JS 表达式。容器 HTML 必须可信；
      // 若来自服务端，不能把未经转义的用户内容拼入模板。
      component.template = container.innerHTML
      // Vue 2 允许在 mount 容器自身声明模板指令；compat 检测并提示迁移。
      if (__COMPAT__ && __DEV__ && container.nodeType === 1) {
        for (let i = 0; i < (container as Element).attributes.length; i++) {
          const attr = (container as Element).attributes[i]
          if (attr.name !== 'v-cloak' && /^(?:v-|:|@)/.test(attr.name)) {
            compatUtils.warnDeprecation(
              DeprecationTypes.GLOBAL_MOUNT_CONTAINER,
              null,
            )
            break
          }
        }
      }
    }

    // 只清空 Element；保留 ShadowRoot 已有节点，包括 Custom Element 在 mount 前注入的 style。
    if (container.nodeType === 1) {
      container.textContent = ''
    }
    const proxy = mount(container, false, resolveRootNamespace(container))
    if (container instanceof Element) {
      container.removeAttribute('v-cloak')
      container.setAttribute('data-v-app', '')
    }
    return proxy
  }

  return app
}) as CreateAppFunction<Element>

第八步：queueJob 为什么能合并更新

文件 / 符号：packages/runtime-core/src/scheduler.ts · queueJob/queueFlush。 输入：带组件 uid 的 instance.job。 状态：queue 增加任务，job 加 QUEUED，currentFlushPromise 从 null 变 Promise。关键代码如下： 

export function queueJob(job: SchedulerJob): void {
  if (!(job.flags! & SchedulerJobFlags.QUEUED)) {
    const jobId = getId(job)
    const lastJob = queue[queue.length - 1]
    if (
      !lastJob ||
      (!(job.flags! & SchedulerJobFlags.PRE) && jobId >= getId(lastJob))
    ) {
      queue.push(job)
    } else {
      queue.splice(findInsertionIndex(jobId), 0, job)
    }

    job.flags! |= SchedulerJobFlags.QUEUED

    queueFlush()
  }
}

function queueFlush() {
  if (!currentFlushPromise) {
    currentFlushPromise = resolvedPromise.then(flushJobs)
  }
}

同一同步栈里再写两次 count，第一次已把 job 标成 QUEUED；后两次在最外层 if 处退出。父组件 uid 通常更小，排序还能让父先更新；若父更新时卸载子组件，子 job 会被标记 DISPOSED，出队时跳过。

微任务到来：job 先 runIfDirty

flushJobs 逐项调用 job；组件 job 正是 effect.runIfDirty.bind(effect)，所以此时才执行 isDirty → effect.run → componentUpdateFn。下面只省略开发 check 的初始化，主 try/finally 与异常清理完整保留： 

function flushJobs(seen?: CountMap) {
  // ... 省略 __DEV__ seen 与 check 初始化

  try {
    for (flushIndex = 0; flushIndex < queue.length; flushIndex++) {
      const job = queue[flushIndex]
      if (job && !(job.flags! & SchedulerJobFlags.DISPOSED)) {
        if (__DEV__ && check(job)) {
          continue
        }
        if (job.flags! & SchedulerJobFlags.ALLOW_RECURSE) {
          job.flags! &= ~SchedulerJobFlags.QUEUED
        }
        callWithErrorHandling(
          job,
          job.i,
          job.i ? ErrorCodes.COMPONENT_UPDATE : ErrorCodes.SCHEDULER,
        )
        if (!(job.flags! & SchedulerJobFlags.ALLOW_RECURSE)) {
          job.flags! &= ~SchedulerJobFlags.QUEUED
        }
      }
    }
  } finally {
    for (; flushIndex < queue.length; flushIndex++) {
      const job = queue[flushIndex]
      if (job) {
        job.flags! &= ~SchedulerJobFlags.QUEUED
      }
    }

    flushIndex = -1
    queue.length = 0

    flushPostFlushCbs(seen)

    currentFlushPromise = null
    if (queue.length || pendingPostFlushCbs.length) {
      flushJobs(seen)
    }
  }
}

callWithErrorHandling 把组件更新异常送入组件错误管道；主 job 即使继续外抛， finally 仍清剩余 QUEUED、清主队列并调用 post flush。 post flush 正常返回后先释放 currentFlushPromise，再递归处理新增任务；原刷新 Promise 在递归返回后完成。裸 post 回调若继续外抛，会中断 post 状态与 Promise 清理。

第十一步：updated 为什么早于 nextTick

patch 完成后，componentUpdateFn 用 queuePostRenderEffect(u, parentSuspense) 登记 onUpdated。随后控制权回到 flushJobs.finally：

  1. 清主队列与 flags；
  2. flushPostFlushCbs(seen)，所以 onUpdated 此时读到 DOM=1；
  3. 若 post 又产生主任务，递归清空；
  4. flushJobs 返回，currentFlushPromise 才兑现；
  5. await nextTick() 的 continuation 才进入后续 Promise reaction。

packages/runtime-core/src/scheduler.ts · nextTick 如下： 

export function nextTick<T, R>(
  this: T,
  fn?: (this: T) => R | Promise<R>,
): Promise<void | R> {
  const p = currentFlushPromise || resolvedPromise
  return fn ? p.then(this ? fn.bind(this) : fn) : p
}
nextTick 保证本轮 Vue 主队列与 post 队列已收敛；不保证浏览器已经 paint，也不等待 fetch、setTimeout 或任意业务 Promise。

  // 其余分支才执行 beforeLeave、leave，并由完成回调触发 remove
}

延迟删除发生在另一种条件下：Transition 自身仍挂载，只用 v-if 等操作移除它的子节点。此时 isUnmounting 为 false， leave 可等动画完成后再调用 performRemove； 组件的逻辑卸载与 post 回调不负责等待这个动画。两种场景要分别追踪。

12.5 测试分别锁定应用清理与生命周期时点

runtime-core 的应用测试先验证未挂载时会警告，再挂载、卸载，并断言宿主容器 已空： 

  // 未挂载时 unmount 会警告；挂载后 unmount 清空根组件的宿主输出。
  test('unmount', () => {
    const Comp = defineComponent({
      props: {
        count: {
          default: 0,
        },
      },
      /**
       * 返回渲染闭包，执行时产出 `props.count`。
       */
      setup(props) {
        return () => props.count
      },
    })

    const root = nodeOps.createElement('div')
    const app = createApp(Comp)

    // 尚未挂载，卸载调用应给出警告。

    app.unmount()

把挂载、更新与卸载连起来

runtime-dom 先把 nodeOps 与 patchProp 注入 runtime-core renderer； renderer 被模块缓存，createAppAPI 则为 Counter 创建独立 app。初次 mount 建立 render effect 并直接运行：render 读 ref、建立依赖， patch(null, subTree) 创建 button，runtime-dom 绑定事件并插入 DOM。 点击时，RefImpl 同步改值并通过 Dep 通知 render effect；reactivity batch 结束后，effect.scheduler 把 runIfDirty job 放入 runtime-core 微任务队列。flushJobs 出队后重新 render，patch 复用原 button，修改文字和 aria；updated 在 post flush 看见新 DOM，整轮 flush 返回后 nextTick 才恢复。app.unmount 再沿闭包 反向进入 render(null)：停止 scope、把 job 标为 DISPOSED、卸载 subTree 并删除 button，最后执行 unmounted 并清理容器标记。

继续读时按这条主线拆章： 组件与 render effect → scheduler → patch → DOM 平台层。列表增删移动再进入 keyed diff；它不是这个文本计数器主线的必经分支。
